# 多会话管理 — 最终方案确认与实现

汇总一下本期范围：

| 功能 | 方案 |
|---|---|
| 会话列表 | 分页加载，按最后活跃时间倒序 |
| 新建会话 | 标准流程 |
| 重命名会话 | 手动修改标题（复用你已有的自动生成逻辑，新增可编辑能力）|
| 归档会话 | 状态字段标记，可恢复 |
| 删除会话 | **硬删除**，不可恢复，需要二次确认交互 |
| 置顶/搜索 | 不做 |
| 多标签同步/会话绑模型 | 不做 |

---

## 一、表结构设计

```sql
CREATE TABLE conversations (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL,
    title VARCHAR(255) NOT NULL DEFAULT '新对话',
    status ENUM('active', 'archived') NOT NULL DEFAULT 'active',
    last_message_at DATETIME DEFAULT NULL,    -- 排序用，每次有新消息就更新
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_status_time (user_id, status, last_message_at DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

**关键索引说明**：`idx_user_status_time` 是复合索引，覆盖了列表查询最常见的过滤条件（某用户 + 某状态 + 按时间排序），能让分页查询直接走索引，避免全表扫描——这是解决你提到的"查询压力大"问题的核心手段。

**关于硬删除的处理**：由于是硬删除，需要考虑级联清理——删除会话时，`messages` 表和 `context_summaries` 表里对应 `conversation_id` 的数据也要一并清理，否则会留下孤儿数据。用数据库外键级联删除，或者应用层显式删除都可以，下面代码里用应用层显式事务处理，更可控。

---

## 二、分页策略选择：游标分页

考虑到你提到"查询压力大"是核心顾虑，**这里直接用游标分页**，性能更稳定，也更适合会话列表这种"下拉加载更多"的典型交互模式。

### 3.2 游标编码/解码工具

```python
# utils/cursor.py
import base64
import json
from datetime import datetime
from typing import Optional, Tuple


def encode_cursor(last_message_at: datetime, conversation_id: int) -> str:
    """把 (时间, id) 组合编码成一个不透明字符串给前端，前端不需要理解内部结构"""
    payload = {"t": last_message_at.isoformat(), "id": conversation_id}
    raw = json.dumps(payload).encode()
    return base64.urlsafe_b64encode(raw).decode()


def decode_cursor(cursor: str) -> Optional[Tuple[datetime, int]]:
    try:
        raw = base64.urlsafe_b64decode(cursor.encode())
        payload = json.loads(raw)
        return datetime.fromisoformat(payload["t"]), payload["id"]
    except Exception:
        return None
```

### 3.3 数据访问层（Repository）

```python
# repository/conversation_repo.py
from typing import Optional, List
from datetime import datetime
from sqlalchemy import text
from utils.cursor import encode_cursor, decode_cursor


class ConversationRepository:
    def __init__(self, db_session_factory):
        self.db_session_factory = db_session_factory

    async def create(self, user_id: int, title: str) -> dict:
        async with self.db_session_factory() as session:
            result = await session.execute(
                text("""
                    INSERT INTO conversations (user_id, title, status, last_message_at)
                    VALUES (:uid, :title, 'active', NOW())
                """),
                {"uid": user_id, "title": title}
            )
            await session.commit()
            new_id = result.lastrowid
            return await self._get_by_id(session, new_id)

    async def list_paginated(
        self,
        user_id: int,
        status: str = "active",
        cursor: Optional[str] = None,
        page_size: int = 20,
    ) -> dict:
        """游标分页核心逻辑：用 (last_message_at, id) 复合条件定位下一页起点"""
        async with self.db_session_factory() as session:
            if cursor:
                decoded = decode_cursor(cursor)
                if not decoded:
                    raise ValueError("无效的分页游标")
                cursor_time, cursor_id = decoded
                sql = text("""
                    SELECT id, title, status, last_message_at, created_at
                    FROM conversations
                    WHERE user_id = :uid AND status = :status
                      AND (last_message_at, id) < (:ctime, :cid)
                    ORDER BY last_message_at DESC, id DESC
                    LIMIT :limit
                """)
                params = {
                    "uid": user_id, "status": status,
                    "ctime": cursor_time, "cid": cursor_id,
                    "limit": page_size + 1,  # 多取一条用于判断是否还有下一页
                }
            else:
                sql = text("""
                    SELECT id, title, status, last_message_at, created_at
                    FROM conversations
                    WHERE user_id = :uid AND status = :status
                    ORDER BY last_message_at DESC, id DESC
                    LIMIT :limit
                """)
                params = {"uid": user_id, "status": status, "limit": page_size + 1}

            result = await session.execute(sql, params)
            rows = [dict(r._mapping) for r in result.fetchall()]

            has_more = len(rows) > page_size
            items = rows[:page_size]

            next_cursor = None
            if has_more and items:
                last_item = items[-1]
                next_cursor = encode_cursor(last_item["last_message_at"], last_item["id"])

            return {"items": items, "next_cursor": next_cursor, "has_more": has_more}

    async def rename(self, user_id: int, conversation_id: int, new_title: str) -> bool:
        async with self.db_session_factory() as session:
            result = await session.execute(
                text("""
                    UPDATE conversations SET title = :title
                    WHERE id = :cid AND user_id = :uid
                """),
                {"title": new_title.strip()[:255], "cid": conversation_id, "uid": user_id}
            )
            await session.commit()
            return result.rowcount > 0

    async def archive(self, user_id: int, conversation_id: int, archived: bool = True) -> bool:
        status = "archived" if archived else "active"
        async with self.db_session_factory() as session:
            result = await session.execute(
                text("UPDATE conversations SET status = :status WHERE id = :cid AND user_id = :uid"),
                {"status": status, "cid": conversation_id, "uid": user_id}
            )
            await session.commit()
            return result.rowcount > 0

    async def hard_delete(self, user_id: int, conversation_id: int) -> bool:
        """
        硬删除：先校验归属权，再级联删除关联数据，最后删主记录
        用事务保证要么全删成功，要么全部回滚，不留孤儿数据
        """
        async with self.db_session_factory() as session:
            async with session.begin():
                owner_check = await session.execute(
                    text("SELECT id FROM conversations WHERE id = :cid AND user_id = :uid"),
                    {"cid": conversation_id, "uid": user_id}
                )
                if not owner_check.first():
                    return False

                # 级联删除关联表数据（按你实际的表清单调整）
                await session.execute(
                    text("DELETE FROM context_summaries WHERE conversation_id = :cid"),
                    {"cid": conversation_id}
                )
                await session.execute(
                    text("DELETE FROM messages WHERE conversation_id = :cid"),
                    {"cid": conversation_id}
                )
                await session.execute(
                    text("DELETE FROM conversations WHERE id = :cid"),
                    {"cid": conversation_id}
                )
            return True

    async def _get_by_id(self, session, conversation_id: int) -> dict:
        result = await session.execute(
            text("""
                SELECT id, title, status, last_message_at, created_at
                FROM conversations WHERE id = :cid
            """),
            {"cid": conversation_id}
        )
        row = result.first()
        return dict(row._mapping) if row else None
```

**关于级联删除的说明**：这里用**应用层显式事务**而不是数据库外键 `ON DELETE CASCADE`，是为了让删除逻辑对开发者可见、可控、可加日志埋点（比如后续想在删除时记录审计日志，这里直接加一行代码就行）。如果你的表已经设置了外键级联，这段代码可以简化，但显式写出来对排查问题更友好。

```python
# routers/conversation.py
from fastapi import APIRouter, HTTPException, Query, Depends
from schemas.conversation import (
    ConversationListResponse, CreateConversationRequest,
    RenameConversationRequest, ConversationActionResponse
)
from repository.conversation_repo import ConversationRepository

router = APIRouter(prefix="/api/conversations", tags=["conversations"])


def get_repo() -> ConversationRepository:
    # 按你项目实际的依赖注入方式替换
    from db import db_session_factory
    return ConversationRepository(db_session_factory)


@router.get("", response_model=ConversationListResponse)
async def list_conversations(
    status: str = Query("active", pattern="^(active|archived)$"),
    cursor: str | None = Query(None),
    page_size: int = Query(20, ge=1, le=50),
    user_id: int = Depends(get_current_user_id),
    repo: ConversationRepository = Depends(get_repo),
):
    try:
        result = await repo.list_paginated(user_id, status, cursor, page_size)
        return result
    except ValueError as e:
        raise HTTPException(400, str(e))


@router.post("", response_model=dict)
async def create_conversation(
    req: CreateConversationRequest,
    user_id: int = Depends(get_current_user_id),
    repo: ConversationRepository = Depends(get_repo),
):
    conv = await repo.create(user_id, req.title)
    return conv


@router.patch("/{conversation_id}/rename", response_model=ConversationActionResponse)
async def rename_conversation(
    conversation_id: int,
    req: RenameConversationRequest,
    user_id: int = Depends(get_current_user_id),
    repo: ConversationRepository = Depends(get_repo),
):
    title = req.title.strip()
    if not title:
        raise HTTPException(400, "标题不能为空")
    ok = await repo.rename(user_id, conversation_id, title)
    if not ok:
        raise HTTPException(404, "会话不存在或无权限")
    return {"status": "ok"}


@router.patch("/{conversation_id}/archive", response_model=ConversationActionResponse)
async def archive_conversation(
    conversation_id: int,
    archived: bool = Query(True),
    user_id: int = Depends(get_current_user_id),
    repo: ConversationRepository = Depends(get_repo),
):
    ok = await repo.archive(user_id, conversation_id, archived)
    if not ok:
        raise HTTPException(404, "会话不存在或无权限")
    return {"status": "ok", "message": "已归档" if archived else "已恢复"}


@router.delete("/{conversation_id}", response_model=ConversationActionResponse)
async def delete_conversation(
    conversation_id: int,
    user_id: int = Depends(get_current_user_id),
    repo: ConversationRepository = Depends(get_repo),
):
    ok = await repo.hard_delete(user_id, conversation_id)
    if not ok:
        raise HTTPException(404, "会话不存在或无权限")
    return {"status": "ok", "message": "已永久删除"}
```

```javascript
  const d = new Date(isoStr);
  const now = new Date();
  const isToday = d.toDateString() === now.toDateString();
  return isToday

function startRename(id, event) {
  event.stopPropagation();
  const titleEl = document.getElementById(`title-${id}`);
  const oldTitle = titleEl.textContent;
  titleEl.classList.add("editing");
```

**关于删除的二次确认交互**：让用户输入 `DELETE` 才能点击确认按钮，这是业界常见的**高危操作防误触设计**（GitHub删除仓库也是这个模式），因为硬删除不可恢复，值得多一步摩擦力防止手滑。

## 五、需要跟你对接确认的两个实现细节
1. **`last_message_at` 字段的更新时机**：这个字段需要在每次该会话产生新消息时更新（无论是用户发的还是模型回复的），建议放在你现有的"消息保存"逻辑里顺带一起 `UPDATE conversations SET last_message_at = NOW() WHERE id = ...`，这样列表排序才能准确反映"最近活跃"。这块需要你在现有的消息写入代码里加这一行。
