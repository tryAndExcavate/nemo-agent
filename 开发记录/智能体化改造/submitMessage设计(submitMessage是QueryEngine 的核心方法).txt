submitMessage(user_input)
    │
    ├─ ① _preprocess_input()   → 解析命令/附件/Memory，得到"处理后的输入"
    ├─ ② _build_message_list() → 拼出本轮要发给模型的完整消息列表
    ├─ ③ query()               → 核心循环（占位，你后续补充）
    ├─ ④ _postprocess()        → 记录用量、存会话、触发 hooks、检查压缩
    └─ ⑤ yield StreamEvent     → 把结果流式抛给调用方

from __future__ import annotations

import asyncio
import json
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, AsyncIterator, Optional


# =========================================================
# 数据结构定义
# =========================================================

class StreamEventType(str, Enum):
    TEXT_DELTA = "text_delta"
    TOOL_USE_START = "tool_use_start"
    TOOL_RESULT = "tool_result"
    MESSAGE_COMPLETE = "message_complete"
    TURN_LIMIT_EXCEEDED = "turn_limit_exceeded"
    BUDGET_EXCEEDED = "budget_exceeded"
    SYSTEM_INFO = "system_info"
    ERROR = "error"
    DONE = "done"


@dataclass
class StreamEvent:
    type: StreamEventType
    data: Any = None
    timestamp: float = field(default_factory=time.time)


@dataclass
class Message:
    """
    贴近 OpenAI 原生消息格式，减少内部转换开销。
    - role: "system" | "user" | "assistant" | "tool"
    - content: str 或 None（assistant 只有 tool_calls 时可为 None）或 list（多模态内容）
    - tool_calls: 仅 assistant 消息可能带，格式与 OpenAI API 一致
    - tool_call_id: 仅 role="tool" 的消息需要，标识对应哪次工具调用
    """
    role: str
    content: Any = None
    tool_calls: Optional[list[dict]] = None
    tool_call_id: Optional[str] = None


@dataclass
class ToolUseBlock:
    """一次工具调用请求（从 OpenAI 流式 delta.tool_calls 累积还原而来）"""
    id: str
    name: str
    input: dict


@dataclass
class ToolResultBlock:
    """一次工具调用的执行结果"""
    tool_use_id: str
    content: Any
    is_error: bool = False


@dataclass
class ProcessedInput:
    text: str
    slash_command: Optional[str] = None
    slash_command_args: Optional[str] = None
    attachments: list[dict] = field(default_factory=list)
    injected_memory: Optional[str] = None
    is_command_only: bool = False


# =========================================================
# QueryEngineConfig（新增 api_key / base_url 字段以支持自定义端点）
# =========================================================

@dataclass
class QueryEngineConfig:
    cwd: str
    tools: list[Any]
    can_use_tool: Any = None
    get_app_state: Any = None
    set_app_state: Any = None
    initial_messages: list[Message] = field(default_factory=list)

    # --- OpenAI 兼容相关配置 ---
    api_key: Optional[str] = None          # 若为 None，走环境变量 OPENAI_API_KEY
    base_url: Optional[str] = None         # 自定义端点，如本地 vLLM / DeepSeek / Kimi 等
    user_specified_model: str = "gpt-4o"

    max_turns: int = 25
    custom_system_prompt: Optional[str] = None
    append_system_prompt: Optional[str] = None
    max_budget_usd: Optional[float] = None
    task_budget: Optional[dict] = None
    commands: list[Any] = field(default_factory=list)
    mcp_clients: list[Any] = field(default_factory=list)
    agents: list[Any] = field(default_factory=list)


# =========================================================
# QueryEngine
# =========================================================

class QueryEngine:
    def __init__(self, config: QueryEngineConfig):
        self.config = config
        self.mutable_messages: list[Message] = list(config.initial_messages)
        self.abort_event: asyncio.Event = asyncio.Event()
        self.permission_denials: list[dict] = []
        self.total_usage: dict = {
            "input_tokens": 0,
            "output_tokens": 0,
            "cost_usd": 0.0,
        }
        self.discovered_skill_names: set[str] = set()
        self.loaded_nested_memory_paths: set[str] = set()

    # =====================================================
    # submitMessage —— 处理一次用户输入
    # =====================================================
    async def submit_message(self, user_input: str) -> AsyncIterator[StreamEvent]:
        try:
            # ① 预处理用户输入
            processed = await self._preprocess_input(user_input)

            if processed.is_command_only:
                async for event in self._handle_command_only(processed):
                    yield event
                yield StreamEvent(StreamEventType.DONE)
                return

            # ② 构建消息列表
            messages = await self._build_message_list(processed)

            system_prompt = self._build_system_prompt()
            tool_schemas = self._build_tool_schemas()

            # ③ 调用 query() 核心循环
            async for raw_event in self.query(messages, system_prompt, tool_schemas):
                # ④ 后处理
                await self._postprocess(raw_event)

                # ⑤ 流式 yield 给调用方
                yield raw_event

                if raw_event.type in (
                    StreamEventType.TURN_LIMIT_EXCEEDED,
                    StreamEventType.BUDGET_EXCEEDED,
                ):
                    break

                if self.abort_event.is_set():
                    yield StreamEvent(StreamEventType.SYSTEM_INFO, {"msg": "已中断"})
                    break

            await self._finalize_turn()
            yield StreamEvent(StreamEventType.DONE)

        except Exception as e:
            yield StreamEvent(StreamEventType.ERROR, {"error": str(e)})
            yield StreamEvent(StreamEventType.DONE)

    # =====================================================
    # ③ query() —— 核心执行循环
    # =====================================================
    async def query(
        self,
        messages: list[Message],
        system_prompt: str,
        tool_schemas: list[dict],
    ) -> AsyncIterator[StreamEvent]:
        current_messages = list(messages)  # 不可变追加：始终基于新 list 操作
        turn_count = 0

        while True:
            turn_count += 1

            # --- 检查轮次限制 ---
            if turn_count > (self.config.max_turns or 25):
                self.mutable_messages = current_messages
                yield StreamEvent(StreamEventType.TURN_LIMIT_EXCEEDED, {"turn": turn_count})
                return

            # --- 调用 LLM API（流式，OpenAI 兼容格式） ---
            accumulated_text = ""
            tool_calls: list[ToolUseBlock] = []
            usage: Optional[dict] = None

            async for chunk in self._call_llm_api_stream(
                messages=current_messages,
                system_prompt=system_prompt,
                tools=tool_schemas,
            ):
                if chunk["type"] == "text":
                    accumulated_text += chunk["text"]
                    yield StreamEvent(StreamEventType.TEXT_DELTA, chunk["text"])

                elif chunk["type"] == "thinking":
                    # 思考块（部分推理模型如 DeepSeek-R1 通过 reasoning_content 返回）
                    # 内部处理，不输出给用户
                    self._handle_thinking_block(chunk)

                elif chunk["type"] == "tool_use":
                    tool_calls.append(
                        ToolUseBlock(id=chunk["id"], name=chunk["name"], input=chunk["input"])
                    )
                    yield StreamEvent(
                        StreamEventType.TOOL_USE_START,
                        {"id": chunk["id"], "name": chunk["name"], "input": chunk["input"]},
                    )

                elif chunk["type"] == "usage":
                    usage = chunk["usage"]

            # --- 组装本轮 assistant 消息（OpenAI 格式：content + tool_calls） ---
            assistant_message = Message(
                role="assistant",
                content=accumulated_text if accumulated_text else None,
                tool_calls=[
                    {
                        "id": tu.id,
                        "type": "function",
                        "function": {
                            "name": tu.name,
                            "arguments": json.dumps(tu.input, ensure_ascii=False),
                        },
                    }
                    for tu in tool_calls
                ] if tool_calls else None,
            )
            current_messages = current_messages + [assistant_message]

            yield StreamEvent(
                StreamEventType.MESSAGE_COMPLETE,
                {"turn": turn_count, "usage": usage},
            )

            # --- 有工具调用？否 → 结束 ---
            if not tool_calls:
                self.mutable_messages = current_messages
                return

            # --- 是 → 并行执行工具调用 ---
            tool_result_blocks = await self._run_tools_parallel(tool_calls)

            for block in tool_result_blocks:
                yield StreamEvent(
                    StreamEventType.TOOL_RESULT,
                    {
                        "tool_use_id": block.tool_use_id,
                        "content": block.content,
                        "is_error": block.is_error,
                    },
                )

            # --- 工具结果追加到消息列表 ---
            # 注意：OpenAI 要求每个工具结果单独一条 role="tool" 消息，
            # 不能像 Anthropic 那样合并进一条 user 消息的 content 数组
            tool_result_messages = [
                Message(
                    role="tool",
                    tool_call_id=b.tool_use_id,
                    content=self._stringify_tool_content(b.content, b.is_error),
                )
                for b in tool_result_blocks
            ]
            current_messages = current_messages + tool_result_messages

            # --- 检查 token 预算 ---
            if self._check_token_budget(current_messages):
                self.mutable_messages = current_messages
                yield StreamEvent(StreamEventType.BUDGET_EXCEEDED, {"turn": turn_count})
                return

            # 循环继续

    # =====================================================
    # query() 的辅助方法（OpenAI 兼容核心逻辑）
    # =====================================================
    async def _call_llm_api_stream(
        self, messages: list[Message], system_prompt: str, tools: list[dict]
    ) -> AsyncIterator[dict]:
        """
        封装 OpenAI 兼容的 Chat Completions 流式调用，
        统一转换成内部 chunk 格式：{"type": "text"|"thinking"|"tool_use"|"usage", ...}
        """
        from openai import AsyncOpenAI

        client = AsyncOpenAI(
            api_key=self.config.api_key,      # None 时会自动读环境变量 OPENAI_API_KEY
            base_url=self.config.base_url,    # 自定义端点，兼容 vLLM/DeepSeek/Kimi/Ollama 等
        )
        model = self.config.user_specified_model or "gpt-4o"

        api_messages = self._to_api_messages(messages, system_prompt)

        stream = await client.chat.completions.create(
            model=model,
            messages=api_messages,
            tools=tools if tools else None,
            stream=True,
            stream_options={"include_usage": True},  # 让最后一个 chunk 带上 usage
        )

        # OpenAI 流式返回的 tool_calls 是按 index 分片传输的，
        # 参数（arguments）是逐段拼接的 JSON 字符串，需要手动累积
        tool_call_accumulator: dict[int, dict] = {}

        async for chunk in stream:
            # 最后一个 usage-only chunk 通常 choices 为空
            if not chunk.choices:
                if getattr(chunk, "usage", None):
                    yield {
                        "type": "usage",
                        "usage": {
                            "input_tokens": chunk.usage.prompt_tokens,
                            "output_tokens": chunk.usage.completion_tokens,
                        },
                    }
                continue

            choice = chunk.choices[0]
            delta = choice.delta

            # --- 部分推理模型（如 DeepSeek-R1）通过 reasoning_content 承载思考过程 ---
            reasoning = getattr(delta, "reasoning_content", None)
            if reasoning:
                yield {"type": "thinking", "text": reasoning}

            # --- 正文文本增量 ---
            if delta.content:
                yield {"type": "text", "text": delta.content}

            # --- 工具调用增量，按 index 累积 ---
            if delta.tool_calls:
                for tc_delta in delta.tool_calls:
                    idx = tc_delta.index
                    if idx not in tool_call_accumulator:
                        tool_call_accumulator[idx] = {"id": "", "name": "", "arguments": ""}

                    if tc_delta.id:
                        tool_call_accumulator[idx]["id"] = tc_delta.id
                    if tc_delta.function:
                        if tc_delta.function.name:
                            tool_call_accumulator[idx]["name"] = tc_delta.function.name
                        if tc_delta.function.arguments:
                            tool_call_accumulator[idx]["arguments"] += tc_delta.function.arguments

                    tool_call_accumulator[idx] = tool_call_accumulator[idx]  # 显式回写，清晰起见

            # --- finish_reason 出现时，说明本轮流结束，把累积的 tool_calls 全部产出 ---
            if choice.finish_reason:
                for idx in sorted(tool_call_accumulator.keys()):
                    tc = tool_call_accumulator[idx]
                    try:
                        parsed_input = json.loads(tc["arguments"] or "{}")
                    except json.JSONDecodeError:
                        parsed_input = {}
                    yield {
                        "type": "tool_use",
                        "id": tc["id"],
                        "name": tc["name"],
                        "input": parsed_input,
                    }
                tool_call_accumulator.clear()

    def _to_api_messages(self, messages: list[Message], system_prompt: str) -> list[dict]:
        """把内部 Message 转换成 OpenAI API 需要的 dict 格式，并在最前面插入 system 消息"""
        api_messages: list[dict] = []
        if system_prompt:
            api_messages.append({"role": "system", "content": system_prompt})

        for m in messages:
            d: dict = {"role": m.role}
            if m.content is not None:
                d["content"] = m.content
            elif m.role == "assistant" and m.tool_calls:
                d["content"] = None  # OpenAI 允许纯 tool_calls 消息 content 为 None
            if m.tool_calls:
                d["tool_calls"] = m.tool_calls
            if m.tool_call_id:
                d["tool_call_id"] = m.tool_call_id
            api_messages.append(d)

        return api_messages

    def _stringify_tool_content(self, content: Any, is_error: bool) -> str:
        """OpenAI 的 tool 消息 content 要求是字符串"""
        if isinstance(content, str):
            text = content
        else:
            try:
                text = json.dumps(content, ensure_ascii=False)
            except (TypeError, ValueError):
                text = str(content)
        return f"[ERROR] {text}" if is_error else text

    def _handle_thinking_block(self, chunk: dict) -> None:
        """思考块：内部处理（如记录日志），不透出给最终用户"""
        # TODO: 可选择性地记录到日志系统，用于调试
        pass

    async def _run_tools_parallel(
        self, tool_calls: list[ToolUseBlock]
    ) -> list[ToolResultBlock]:
        """并行执行多个工具调用"""

        async def run_one(tu: ToolUseBlock) -> ToolResultBlock:
            if self.config.can_use_tool and not self.config.can_use_tool(tu.name, tu.input):
                self.permission_denials.append({"tool": tu.name, "input": tu.input})
                return ToolResultBlock(
                    tool_use_id=tu.id,
                    content="Permission denied by user configuration.",
                    is_error=True,
                )

            tool = next((t for t in self.config.tools if t.name == tu.name), None)
            if tool is None:
                return ToolResultBlock(
                    tool_use_id=tu.id,
                    content=f"Tool '{tu.name}' not found.",
                    is_error=True,
                )

            try:
                result = await tool.call(tu.input, self.config.cwd)
                return ToolResultBlock(
                    tool_use_id=tu.id, content=result.content, is_error=result.is_error
                )
            except Exception as e:
                return ToolResultBlock(tool_use_id=tu.id, content=str(e), is_error=True)

        return await asyncio.gather(*(run_one(tu) for tu in tool_calls))

    def _check_token_budget(self, messages: list[Message]) -> bool:
        if self.config.task_budget is None:
            return False
        # TODO: 用真实 tokenizer（如 tiktoken）替代粗略估算
        estimated_tokens = sum(len(str(m.content)) for m in messages) // 4
        return estimated_tokens > self.config.task_budget["total"]

    def _build_system_prompt(self) -> str:
        prompt = self.config.custom_system_prompt or "You are a helpful assistant."
        if self.config.append_system_prompt:
            prompt += "\n" + self.config.append_system_prompt
        return prompt

    def _build_tool_schemas(self) -> list[dict]:
        """OpenAI function calling 格式：{"type": "function", "function": {...}}"""
        return [
            {
                "type": "function",
                "function": {
                    "name": t.name,
                    "description": t.description,
                    "parameters": t.input_schema,
                },
            }
            for t in self.config.tools
        ]

    # =====================================================
    # ① 预处理用户输入
    # =====================================================
    async def _preprocess_input(self, user_input: str) -> ProcessedInput:
        text = user_input.strip()
        slash_command, slash_args, is_command_only = self._parse_slash_command(text)
        if slash_command:
            text = slash_args or ""
        attachments = self._extract_attachments(user_input)
        injected_memory = await self._load_relevant_memory(text)
        return ProcessedInput(
            text=text,
            slash_command=slash_command,
            slash_command_args=slash_args,
            attachments=attachments,
            injected_memory=injected_memory,
            is_command_only=is_command_only,
        )

    def _parse_slash_command(self, text: str) -> tuple[Optional[str], Optional[str], bool]:
        if not text.startswith("/"):
            return None, None, False
        parts = text[1:].split(maxsplit=1)
        command_name = parts[0] if parts else ""
        args = parts[1] if len(parts) > 1 else ""
        matched = next((c for c in self.config.commands if c.name == command_name), None)
        if matched is None:
            return None, None, False
        is_local_only = getattr(matched, "is_local_only", False)
        return command_name, args, is_local_only

    def _extract_attachments(self, user_input: str) -> list[dict]:
        # TODO: 解析 @file.py 引用等，读取内容打包成附件
        return []

    async def _load_relevant_memory(self, text: str) -> Optional[str]:
        # TODO: 结合 self.loaded_nested_memory_paths 做去重加载
        return None

    async def _handle_command_only(self, processed: ProcessedInput) -> AsyncIterator[StreamEvent]:
        yield StreamEvent(
            StreamEventType.SYSTEM_INFO,
            {"msg": f"执行本地命令: {processed.slash_command}"},
        )

    # =====================================================
    # ② 构建消息列表
    # =====================================================
    async def _build_message_list(self, processed: ProcessedInput) -> list[Message]:
        content: Any = processed.text
        if processed.attachments:
            content = self._merge_attachments_into_content(processed.text, processed.attachments)

        new_user_message = Message(role="user", content=content)
        messages = list(self.mutable_messages)
        if processed.injected_memory:
            messages.append(Message(role="user", content=f"[Memory]\n{processed.injected_memory}"))
        messages.append(new_user_message)

        self.mutable_messages = messages
        return messages

    def _merge_attachments_into_content(self, text: str, attachments: list[dict]) -> Any:
        """
        OpenAI 视觉模型的多模态格式：
        [{"type": "text", "text": ...}, {"type": "image_url", "image_url": {"url": ...}}]
        """
        blocks = [{"type": "text", "text": text}]
        blocks.extend(attachments)
        return blocks

    # =====================================================
    # ④ 后处理
    # =====================================================
    async def _postprocess(self, event: StreamEvent) -> None:
        if event.type == StreamEventType.MESSAGE_COMPLETE:
            usage = event.data.get("usage") if event.data else None
            if usage:
                self._record_usage(usage)

        elif event.type == StreamEventType.BUDGET_EXCEEDED:
            self.abort_event.set()

        await self._trigger_hooks(event)

    def _record_usage(self, usage: dict) -> None:
        self.total_usage["input_tokens"] += usage.get("input_tokens", 0)
        self.total_usage["output_tokens"] += usage.get("output_tokens", 0)
        # TODO: 按 model 单价换算 cost_usd
        if self.config.max_budget_usd and self.total_usage["cost_usd"] >= self.config.max_budget_usd:
            self.abort_event.set()

    async def _trigger_hooks(self, event: StreamEvent) -> None:
        pass

    async def _finalize_turn(self) -> None:
        await self._save_session()
        await self._check_compaction()

    async def _save_session(self) -> None:
        pass

    async def _check_compaction(self) -> None:
        pass

    def abort(self) -> None:
        self.abort_event.set()
