借鉴Claude Code中QueryEngine的设计，来实现我们的Agent引擎

完整对话的全生命周期：
1.维护消息历史
2.调用ApI
3.执行工具调用
4.管理token预算
5.处理错误和重试
6.触发上下文压缩

Python 实现：
# ---------- 基础数据结构 ----------
Role = Literal["user", "assistant", "system", "tool"]

@dataclass
class Message:
    role: Role
    content: Any  # str 或 结构化 content blocks（text / tool_use / tool_result）
    id: Optional[str] = None

@dataclass
class ToolResult:
    content: Any
    is_error: bool = False

class Tool(Protocol):
    """工具协议：只要实现这三样，就能被 QueryEngine 调用"""
    name: str
    description: str
    input_schema: dict  # JSON Schema，用于告诉模型如何调用

    async def call(self, input: dict, cwd: str) -> ToolResult: ...

# ---------- 权限检查 ----------
# 最简单版本：同步函数，返回是否允许执行
CanUseToolFn = Callable[[str, dict], bool]

def allow_all(tool_name: str, tool_input: dict) -> bool:
    return True

# ---------- 应用状态 ----------
@dataclass
class AppState:
    messages: list[Message] = field(default_factory=list)
    total_cost_usd: float = 0.0
    turn_count: int = 0

GetAppState = Callable[[], AppState]
SetAppState = Callable[[Callable[[AppState], AppState]], None]

# ---------- 核心配置 ----------
@dataclass
class QueryEngineConfig:
    # --- P0 必需 ---
    cwd: str
    tools: list[Tool]
    can_use_tool: CanUseToolFn = allow_all
    get_app_state: GetAppState = None
    set_app_state: SetAppState = None
    initial_messages: list[Message] = field(default_factory=list)
    model: str = "claude-sonnet-4-5-20250929"
    max_turns: int = 25

    # --- P1 建议保留 ---
    custom_system_prompt: Optional[str] = None
    append_system_prompt: Optional[str] = None
    max_budget_usd: Optional[float] = None

    # --- P2 先留空占位，后续扩展 ---
    mcp_clients: list[Any] = field(default_factory=list)
    agents: list[Any] = field(default_factory=list)
    commands: list[Any] = field(default_factory=list)


class QueryEngine:
    def __init__(self, config: "QueryEngineConfig"):
        self.config = config
        self.mutable_messages = list(config.initial_messages)
        self.abort_event = asyncio.Event()
        self.permission_denials: list[dict] = []
        self.total_usage = {"input_tokens": 0, "output_tokens": 0, "cost_usd": 0.0}
        self.discovered_skill_names: set[str] = set()
        self.loaded_nested_memory_paths: set[str] = set()

    def abort(self):
        self.abort_event.set()


class QueryEngineRegistry:
    """全局注册表：每个 session_id 对应一个"单例" QueryEngine"""
    _engines: Dict[str, QueryEngine] = {}
    _lock = asyncio.Lock()

    @classmethod
    async def get_or_create(cls, session_id: str, config: "QueryEngineConfig") -> QueryEngine:
        async with cls._lock:
            if session_id not in cls._engines:
                cls._engines[session_id] = QueryEngine(config)
            return cls._engines[session_id]

    @classmethod
    def get(cls, session_id: str) -> Optional[QueryEngine]:
        return cls._engines.get(session_id)

    @classmethod
    def destroy(cls, session_id: str):
        """结束会话时释放资源"""
        engine = cls._engines.pop(session_id, None)
        if engine:
            engine.abort()
