## 二、整体架构图

```
┌───────────────────────────────────────────────────────────┐
│ 主 Agent（QueryEngine, engine_mode=MAIN, conversation_id=C123）│
│  · live memory_store（本次运行期持续写入）                     │
│  · jsonl（历史大轮次持久化，只有主Agent自己写入的内容）             │
│  · 工具集含 dispatch_agent（唯一委派入口）                      │
└──────────────────────┬────────────────────────────────────┘
                       │ dispatch_agent(agent_name="memory-retriever", task="...")
                       ▼
┌───────────────────────────────────────────────────────────┐
│ Dispatcher                                                  │
│  · 校验 ExecutionContext：depth 是否超限、call_path 是否成环   │
│  · 构建 MemoryRetrievalContext（两个只读窗口）                 │
└──────────────────────┬────────────────────────────────────┘
                       ▼
┌───────────────────────────────────────────────────────────┐
│ SubAgentRunner                                              │
│  创建全新 QueryEngine(engine_mode=SUBAGENT,                  │
│                       conversation_id=uuid4新生成)            │
│  persist_conversation=False / persist_memory=False           │
│  tools = 仅 6 个只读记忆工具，不含 dispatch_agent               │
└──────────────────────┬────────────────────────────────────┘
                       │ 内部多轮：分别/交叉查两个来源 → 综合
                       ▼
              LiveMemoryReadOnlyView      ConversationHistoryReader
              （借用主Agent memory_store）  （只读绑定 C123.jsonl，路径已锁死）
                       │                          │
                       └───────────┬──────────────┘
                                   ▼
                          只返回一段简报文字
                       （中间过程 & 临时会话用完即弃，不落盘）
```

### 1. 执行模式与权限策略

```python
# subagent/policy.py
from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum


class EngineMode(Enum):
    MAIN = "main"          # 主Agent：可写记忆、可持久化、可dispatch
    SUBAGENT = "subagent"  # 子Agent：只读、不持久化、默认不可dispatch


@dataclass
class ExecutionBudget:
    max_llm_turns: int = 5
    max_tool_calls: int = 10
    max_memory_detail_reads: int = 3   # 防止子Agent疯狂拉全量详情
    max_dispatch_depth: int = 0        # 0 表示该Agent不能再往下派遣


@dataclass(frozen=True)
class AgentExecutionContext:
    """每次 dispatch 都会派生一个新的 context，用于深度/环检测"""
    root_conversation_id: str
    current_agent: str
    depth: int
    call_path: tuple[str, ...]
    max_depth: int

    def derive(self, next_agent: str, next_max_depth: int) -> "AgentExecutionContext":
        if self.depth + 1 > self.max_depth:
            raise PermissionError(
                f"dispatch depth exceeded: current depth={self.depth}, max={self.max_depth}"
            )
        if next_agent in self.call_path:
            raise PermissionError(
                f"cyclic subagent call detected: {self.call_path} -> {next_agent}"
            )
        return AgentExecutionContext(
            root_conversation_id=self.root_conversation_id,
            current_agent=next_agent,
            depth=self.depth + 1,
            call_path=self.call_path + (next_agent,),
            max_depth=next_max_depth,
        )

    @classmethod
    def root(cls, conversation_id: str, max_depth: int) -> "AgentExecutionContext":
        return cls(
            root_conversation_id=conversation_id,
            current_agent="main",
            depth=0,
            call_path=("main",),
            max_depth=max_depth,
        )
```

要点：
- `depth` / `call_path` 不是"建议"，是每次 dispatch 前必须通过的硬校验。
- `main → memory-retriever → memory-retriever` 会因为 `"memory-retriever" in call_path` 被直接拒绝。
- `main → A → B → C` 超过 `max_depth` 同样会被拒绝，而不是靠 `max_turns` 间接限制。

#### 2.2 历史 JSONL 只读访问器（路径锁死 + 懒加载索引）

```python
# memory/history_reader.py
from __future__ import annotations
import json, re
from pathlib import Path
from dataclasses import dataclass


@dataclass
class HistoryRecord:
    id: str
    task_id: str
    turn: int
    timestamp: float
    summary: str
    keywords: list[str]
    full_messages: list[dict]  # 已序列化的消息，读出来时按需渲染


_SAFE_ID_PATTERN = re.compile(r"[A-Za-z0-9_-]+")


def _resolve_safe_path(memory_root: Path, conversation_id: str) -> Path:
    """
    防止 conversation_id 被拼接出目录穿越，或指向别的用户/会话文件。
    这个函数是唯一允许"翻译" conversation_id -> 文件路径的地方。
    """
    if not _SAFE_ID_PATTERN.fullmatch(conversation_id):
        raise ValueError(f"invalid conversation_id: {conversation_id}")

    root = memory_root.resolve()
    path = (root / f"{conversation_id}.jsonl").resolve()

    if path.parent != root:
        raise PermissionError("resolved history path escaped memory_root")

    return path


class ConversationHistoryReader:
    """
    只读访问某一个 conversation_id 对应的 jsonl 历史文件。
    构造时就把路径锁死，之后任何方法都不接受 conversation_id 参数,
    杜绝"顺手查询别的会话"的可能性。
    """

    def __init__(self, memory_root: Path, conversation_id: str):
        self._path = _resolve_safe_path(memory_root, conversation_id)
        self._cache: list[HistoryRecord] | None = None

    def _load(self) -> list[HistoryRecord]:
        if self._cache is not None:
            return self._cache
        records: list[HistoryRecord] = []
        if self._path.exists():
            with self._path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    obj = json.loads(line)
                    records.append(HistoryRecord(
                        id=obj["id"],
                        task_id=obj.get("task_id", "default"),
                        turn=obj.get("turn", 0),
                        timestamp=obj.get("timestamp", 0.0),
                        summary=obj.get("summary", ""),
                        keywords=obj.get("keywords", []),
                        full_messages=obj.get("full_messages", []),
                    ))
        self._cache = records
        return records

    def list_index(self, task_id: str | None = None) -> list[dict]:
        records = self._load()
        if task_id:
            records = [r for r in records if r.task_id == task_id]
        return [{"id": r.id, "task_id": r.task_id, "turn": r.turn, "summary": r.summary}
                for r in sorted(records, key=lambda x: x.turn)]

    def search(self, query: str, top_k: int = 5) -> list[dict]:
        query_words = set(query.lower().split())
        scored = []
        for r in self._load():
            pool = (r.summary + " " + " ".join(r.keywords)).lower()
            overlap = len(query_words & set(pool.split()))
            if overlap > 0:
                scored.append((overlap, r))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [{"id": r.id, "task_id": r.task_id, "summary": r.summary} for _, r in scored[:top_k]]

    def get_detail(self, record_id: str) -> HistoryRecord | None:
        for r in self._load():
            if r.id == record_id:
                return r
        return None
```

要点：
- `conversation_id` 只在构造函数里出现一次，之后彻底与工具层解耦。
- 即便未来 `search_history` 的 schema 被误改成包含 `conversation_id` 字段，`ConversationHistoryReader` 实例本身也不接受运行时传参改变绑定文件，双保险。

### 3. 子智能体专用工具（两组，语义显式区分）

```python
# tools/memory_agent_tools.py
from tool import Tool, ToolResult, ToolUseContext


def _render_history_messages(full_messages: list[dict]) -> str:
    lines = []
    for m in full_messages:
        role = m.get("role", "unknown")
        content = m.get("content", "")
        lines.append(f"{role}: {content}")
    return "\n".join(lines)


# ---------- current / live memory ----------

class ListCurrentMemoryTool(Tool):
    name = "list_current_memory"
    description = "列出主Agent【当前这次运行】已经积累的记忆目录（仅摘要）。"
    input_schema = {"type": "object", "properties": {}}

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        view = context.memory_retrieval.live_memory
        index = view.list_index()
        if not index:
            return ToolResult(output="当前运行期暂无记忆")
        lines = [f"[{c['id']}] (任务:{c['task_id']}, 第{c['turn']}轮) {c['summary']}" for c in index]
        return ToolResult(output="\n".join(lines))


class SearchCurrentMemoryTool(Tool):
    name = "search_current_memory"
    description = "在【当前运行期】记忆中按关键词搜索。"
    input_schema = {
        "type": "object",
        "properties": {"query": {"type": "string"}},
        "required": ["query"],
    }

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        view = context.memory_retrieval.live_memory
        results = view.search(input["query"])
        if not results:
            return ToolResult(output="当前运行期未找到匹配内容")
        lines = [f"[{r['id']}] (任务:{r['task_id']}) {r['summary']}" for r in results]
        return ToolResult(output="\n".join(lines))


class GetCurrentMemoryDetailTool(Tool):
    name = "get_current_memory_detail"
    description = "获取【当前运行期】某条记忆的完整内容。chunk_id 必须来自之前工具返回的真实ID。"
    input_schema = {
        "type": "object",
        "properties": {"chunk_id": {"type": "string"}},
        "required": ["chunk_id"],
    }

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        context.consume_detail_budget()  # 见下方 ToolUseContext 扩展
        view = context.memory_retrieval.live_memory
        chunk = view.get_detail(input["chunk_id"])
        if chunk is None:
            return ToolResult(output="未找到该记忆片段", is_error=True)
        text = "\n".join(str(m) for m in chunk.full_messages)
        return ToolResult(output=text)


# ---------- historical / jsonl memory ----------

class ListHistoryIndexTool(Tool):
    name = "list_history_index"

class SearchHistoryTool(Tool):
    name = "search_history"
    description = "在【以往大轮次持久化历史】中按关键词搜索。不需要也不能指定conversation。"
    input_schema = {
        "type": "object",
        "properties": {"query": {"type": "string"}},
        "required": ["query"],

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        reader = context.memory_retrieval.history

class GetHistoryDetailTool(Tool):
    name = "get_history_detail"
    description = "获取【历史记录】中某条的完整内容。record_id 必须来自之前工具返回的真实ID。"
    input_schema = {
        "type": "object",
        "properties": {"record_id": {"type": "string"}},
        "required": ["record_id"],
    }

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        context.consume_detail_budget()
        reader = context.memory_retrieval.history
        record = reader.get_detail(input["record_id"])
        if record is None:
            return ToolResult(output="未找到该历史记录", is_error=True)
        return ToolResult(output=_render_history_messages(record.full_messages))
```

注意：schema 里**没有任何一个字段是 `conversation_id`**。这不是遗漏，是刻意设计——凡是安全边界，都不能出现在模型可控的参数里。

### 4. `ToolUseContext` 扩展：记忆访问 + 预算消耗

```python
# tool.py（关键片段）
from dataclasses import dataclass, field
import asyncio


@dataclass
class ToolUseContext:
    cwd: str
    abort_event: asyncio.Event
    messages: tuple
    extras: dict = field(default_factory=dict)

    @property
    def memory_retrieval(self):
        """子Agent专用：拿到的是 MemoryRetrievalContext，而不是原始 store"""
        return self.extras.get("memory_retrieval")

    def consume_detail_budget(self):
        """
        每次调用 get_*_detail 都会消耗一次预算。
        超限直接抛异常中断该轮工具执行，而不是让模型无限拉详情。
        """
        state = self.extras.get("execution_state")
        if state is None:
            return
        state.detail_reads += 1
        if state.detail_reads > state.budget.max_memory_detail_reads:
            raise PermissionError("memory detail read budget exceeded")
```

### 6. Dispatcher：构建双通道记忆 + 硬校验深度/环

```python
# tools/dispatch_agent_tool.py
from tool import Tool, ToolResult, ToolUseContext
from subagent.registry import SubAgentRegistry
from subagent.runner import run_subagent
from memory.live_view import LiveMemoryReadOnlyView
from memory.history_reader import ConversationHistoryReader
from memory.retrieval_context import MemoryRetrievalContext


class DispatchAgentTool(Tool):
    name = "dispatch_agent"

    def __init__(self, allowed_subagents: set[str] | None = None):
        # 允许 Runner 在构造子Agent工具集时传入白名单，物理限制可派遣的对象
        self._allowed = allowed_subagents
        self.input_schema = self._build_schema()
        self.description = self._build_description()

    def _candidates(self):
        all_defs = SubAgentRegistry.list_available()
        if self._allowed is None:
            return all_defs
        return [d for d in all_defs if d.name in self._allowed]

    def _build_description(self) -> str:
        available = self._candidates()
        if not available:
            return "调用一个专门的子智能体处理特定任务（当前无可用子智能体）。"
        lines = ["将任务委派给专门的子智能体处理。\n可用的子智能体："]
        lines += [f"- {d.name}: {d.description}" for d in available]
        return "\n".join(lines)

    def _build_schema(self) -> dict:
        names = [d.name for d in self._candidates()] or ["none"]
        return {
            "type": "object",
            "properties": {
                "agent_name": {"type": "string", "enum": names},
                "task": {"type": "string"},
            },
            "required": ["agent_name", "task"],
        }

    async def execute(self, input: dict, context: ToolUseContext) -> ToolResult:
        definition = SubAgentRegistry.get(input["agent_name"])
        if definition is None or (self._allowed is not None and definition.name not in self._allowed):
            return ToolResult(output=f"未找到或无权调用子智能体: {input['agent_name']}", is_error=True)

        exec_ctx = context.extras.get("execution_context")
        try:
            child_ctx = exec_ctx.derive(
                next_agent=definition.name,
                next_max_depth=definition.budget.max_dispatch_depth,
            )
        except PermissionError as e:
            return ToolResult(output=f"派遣被拒绝: {e}", is_error=True)

        # 按需组装双通道记忆，只有 definition 显式声明需要才给
        retrieval = None
        if definition.needs_live_memory or definition.needs_history_memory:
            retrieval = MemoryRetrievalContext(
                live_memory=(
                    LiveMemoryReadOnlyView(context.memory_store)
                    if definition.needs_live_memory else None
                ),
                history=(
                    ConversationHistoryReader(
                        memory_root=context.extras["memory_root"],
                        conversation_id=exec_ctx.root_conversation_id,  # 永远是根conversation，不可被覆盖
                    ) if definition.needs_history_memory else None
                ),
            )

        result = await run_subagent(
            definition=definition,
            task=input["task"],
            parent_config=context.extras.get("parent_config"),
            execution_context=child_ctx,
            memory_retrieval=retrieval,
        )
        return ToolResult(output=result.text, is_error=result.is_error)
```

关键点：
- `exec_ctx.derive(...)` 是硬校验，`PermissionError` 直接把结果返回给模型（模型能"看到"被拒绝，但拒绝逻辑本身模型改不了）。
- `history` 永远绑定 `exec_ctx.root_conversation_id`，这是从最初 root context 一路传下来的值，不会因为子Agent或更下层Agent传参而改变。

### 7. SubAgentRunner：真正的"独立于其他区域"

```python
# subagent/runner.py
from __future__ import annotations
import uuid
from typing import Optional
from .definition import SubAgentDefinition
from .policy import AgentExecutionContext, EngineMode


class ExecutionState:
    def __init__(self, budget):
        self.budget = budget
        self.detail_reads = 0
        self.tool_calls = 0


class SubAgentResult:
    def __init__(self, text: str, is_error: bool = False, cost_usd: float = 0.0):
        self.text = text
        self.is_error = is_error
        self.cost_usd = cost_usd


async def run_subagent(
    definition: SubAgentDefinition,
    task: str,
    parent_config,
    execution_context: AgentExecutionContext,
    memory_retrieval=None,
) -> SubAgentResult:
    from query_engine import QueryEngine, QueryEngineConfig

    # 独立、全新、用完即弃的 conversation_id —— 与主Agent的C123毫无关系
    sub_conversation_id = f"subagent-{definition.name}-{uuid.uuid4().hex[:8]}"

    exec_state = ExecutionState(definition.budget)

    sub_config = QueryEngineConfig(
        conversation_id=sub_conversation_id,
        engine_mode=EngineMode.SUBAGENT,
        tools=definition.tools_factory(),  # 只有 definition 里声明的工具，不含 dispatch_agent
        api_key=parent_config.api_key,
        base_url=parent_config.base_url,
        user_specified_model=definition.model or parent_config.user_specified_model,
        custom_system_prompt=definition.system_prompt,
        max_turns=definition.budget.max_llm_turns,
        persist_conversation=False,   # 不落盘，不产生垃圾 jsonl
        persist_memory=False,         # 不写任何 memory_store
    )

    engine = QueryEngine(sub_config)
    engine._extra_tool_context = {
        "memory_retrieval": memory_retrieval,
        "execution_context": execution_context,
        "execution_state": exec_state,
        # 注意：不传 memory_root / parent_config，子Agent拿不到组装history reader所需的原料
        # 需要更深层dispatch时，由更上一层的Dispatcher重新组装，而不是让子Agent自己转发
    }

    final_text, is_error = "", False
    try:
        async for event in engine.submit_message(task):
            if event.type.value == "text_delta":
                final_text += event.data
            elif event.type.value == "error":
                is_error = True
                final_text = f"子智能体执行出错: {event.data}"
    except Exception as e:
        is_error = True
        final_text = f"子智能体异常: {e}"

    return SubAgentResult(
        text=final_text.strip(),
        is_error=is_error,
        cost_usd=engine.state.total_cost_usd,
    )
```

要点：
- `persist_conversation=False` / `persist_memory=False` 保证不会产生 `subagent-memory-retriever-xxxx.jsonl` 这类垃圾文件，也不会污染未来的历史检索结果。
- `extra_tool_context` 里刻意不包含 `memory_root` 和 `parent_config`，即便 `memory-retriever` 的 `allow_dispatch` 被错误设为 `True`，它也拼不出合法的 `ConversationHistoryReader`——**这是针对"代码未来被误改"的第二层防御**。

### 8. QueryEngine：按 `engine_mode` 物理裁剪能力

```python
# query_engine.py（关键片段）
from subagent.policy import EngineMode
from subagent.registry import SubAgentRegistry
from tools.dispatch_agent_tool import DispatchAgentTool
from memory_store import MemoryStore


class QueryEngine:
    def __init__(self, config):
        self.config = config
        self.tools = list(config.tools)

        if config.engine_mode == EngineMode.MAIN:
            self.memory_store = MemoryStore()
            self._history_writer = ConversationHistoryWriter(  # 追加写入 C123.jsonl
                memory_root=config.memory_root,
                conversation_id=config.conversation_id,
            )
            if SubAgentRegistry.list_available():
                self.tools.append(
                    DispatchAgentTool(allowed_subagents=config.allowed_subagents)
                )
        else:  # EngineMode.SUBAGENT
            self.memory_store = None
            self._history_writer = None
            # 子Agent的 tools 完全来自 definition.tools_factory()，
            # 这里不追加任何东西，包括 dispatch_agent

        self._persist_conversation = getattr(config, "persist_conversation", True)
        self._persist_memory = getattr(config, "persist_memory", True)

    def _build_tool_use_context(self) -> ToolUseContext:
        extras = dict(self._extra_tool_context) if hasattr(self, "_extra_tool_context") else {}
        extras.setdefault("parent_config", self.config)
        if self.memory_store is not None:
            extras.setdefault("memory_store", self.memory_store)
        return ToolUseContext(
            cwd=get_global_state().cwd,
            abort_event=self.abort_event,
            messages=tuple(self.state.messages),
            extras=extras,
        )

    def _maybe_persist_turn(self, messages_delta: list):
        """只有 MAIN 模式、且 persist_memory=True 才写 live memory + jsonl"""
        if self.memory_store is not None and self._persist_memory:
            chunk = self.memory_store.add_chunk(
                messages_delta,
                summary=self._summarize(messages_delta),
                task_id=self.config.current_task_id,
            )
            if self._persist_conversation and self._history_writer is not None:
                self._history_writer.append(chunk)
```

主 Agent 的历史大轮次持久化，走的是新增的 `ConversationHistoryWriter`（与 `ConversationHistoryReader` 对称，只在 `MAIN` 模式下存在）：

```python
# memory/history_writer.py
import json, time
from pathlib import Path
from memory.history_reader import _resolve_safe_path


class ConversationHistoryWriter:
    """只有主Agent持有，负责把 chunk 追加写入 conversation_id.jsonl"""

    def __init__(self, memory_root: Path, conversation_id: str):
        self._path = _resolve_safe_path(memory_root, conversation_id)
        self._path.parent.mkdir(parents=True, exist_ok=True)

    def append(self, chunk) -> None:
        record = {
            "id": chunk.id,
            "task_id": chunk.task_id,
            "turn": chunk.turn,
            "timestamp": chunk.timestamp,
            "summary": chunk.summary,
            "keywords": list(chunk.keywords),
            "full_messages": [self._serialize(m) for m in chunk.full_messages],
        }
        with self._path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")

    def _serialize(self, msg) -> dict:
        return {"role": getattr(msg, "role", "unknown"), "content": str(getattr(msg, "content", ""))}
```

## 四、权限边界总表

| 维度 | 主 Agent（MAIN） | memory-retriever（SUBAGENT） |
|---|---|---|
| `conversation_id` | 固定，如 `C123` | 每次全新 `uuid4`，用完即弃 |
| `messages` | 持续累积 | 空开始，不继承主Agent历史 |
| live memory 读 | ✅ | ✅（只读视图） |
| live memory 写 | ✅ | ❌（拿不到 `add_chunk`） |
| `C123.jsonl` 读 | ✅ | ✅（路径已锁死，只读） |
| `C123.jsonl` 写 | ✅（`ConversationHistoryWriter`） | ❌（没有该对象） |
| 自身对话持久化 | ✅ | ❌（`persist_conversation=False`） |
| `dispatch_agent` 工具 | ✅ | ❌（工具集里物理不存在） |
| 可指定 `conversation_id` 参数 | 不适用 | ❌（任何工具 schema 都没有这个字段） |
| 调用深度/环检测 | `depth=0` 起点 | `derive()` 时强制校验，`max_dispatch_depth=0` |
| 执行预算 | `max_turns`（配置项） | `max_llm_turns` / `max_tool_calls` / `max_memory_detail_reads` 三重限制 |
