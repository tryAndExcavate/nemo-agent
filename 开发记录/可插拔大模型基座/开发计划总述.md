# 可插拔大模型基座 —— 架构设计与实现

这个需求本质上要解决两个问题：

1. **UI 层**：可视化地管理"哪些模型可用、当前用哪个"，操作体验类似工作流节点编辑器（基座节点 = 容器，模型 = 可插拔的插件）。
2. **工程层**：真正让后端调用大模型的代码"感知不到"底层是 OpenAI、Claude、通义千问还是自建接口——这需要一个**适配器模式（Adapter Pattern）**，而不仅仅是把配置存成 JSON。

下面按 **数据结构 → 后端(FastAPI) → 前端(HTML/JS) → 调用侧如何真正"可插拔"** 四部分给出完整代码。

---

## 一、数据结构设计（JSON 存储）

```json
{
  "active_model_id": "b3f1a2c4-xxxx",
  "models": [
    {
      "id": "b3f1a2c4-xxxx",
      "name": "线上-GPT4turbo",
      "provider": "openai",
      "api_base": "https://api.openai.com/v1",
      "api_key": "sk-xxxxxx",
      "model_name": "gpt-4-turbo",
      "temperature": 0.7,
      "max_tokens": 4096,
      "extra_params": {}
    }
  ]
}
```

- `provider` 决定用哪个适配器（openai / azure_openai / anthropic / zhipu / qwen / custom...）
- `model_name` 是平台方真正调用时用的模型标识，以后平台改名字只需要改这一个字段，不用动代码
- `active_model_id` 就是"插在基座上的那个模型"

## 二、后端：FastAPI + JSON 存储

```python
# main.py
import json
import os
import uuid
from typing import Optional, Dict, Any, List

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

app = FastAPI(title="可插拔大模型基座")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

DATA_FILE = "models.json"


# ---------- 数据模型 ----------
class ModelConfig(BaseModel):
    id: Optional[str] = None
    name: str
    provider: str
    api_base: str
    api_key: str
    model_name: str
    temperature: Optional[float] = 0.7
    max_tokens: Optional[int] = 4096
    extra_params: Optional[Dict[str, Any]] = {}


class SaveState(BaseModel):
    active_model_id: Optional[str] = None
    models: List[ModelConfig]


# ---------- 文件读写 ----------
def load_data() -> dict:
    if not os.path.exists(DATA_FILE):
        return {"active_model_id": None, "models": []}
    with open(DATA_FILE, "r", encoding="utf-8") as f:
        return json.load(f)


def save_data(data: dict):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)


# ---------- 接口 ----------
@app.get("/api/models")
def get_models():
    """页面加载时拉取全部模型 + 当前激活的模型"""
    return load_data()


@app.post("/api/models/save")
def save_models(state: SaveState):
    """
    前端点击"保存"时，把整个状态（模型列表 + 当前激活id）
    一次性写入 json。没有 id 的新模型自动生成 uuid。
    """
    data = state.dict()
    for m in data["models"]:
        if not m.get("id"):
            m["id"] = str(uuid.uuid4())

    ids = {m["id"] for m in data["models"]}
    if data.get("active_model_id") and data["active_model_id"] not in ids:
        raise HTTPException(400, "激活的模型不在模型列表中")

    save_data(data)
    return {"status": "ok", "data": data}


@app.delete("/api/models/{model_id}")
def delete_model(model_id: str):
    data = load_data()
    data["models"] = [m for m in data["models"] if m["id"] != model_id]
    if data.get("active_model_id") == model_id:
        data["active_model_id"] = None
    save_data(data)
    return {"status": "ok"}


@app.get("/api/models/active")
def get_active_model():
    """真正调用大模型时，agent 代码只依赖这个接口/函数"""
    data = load_data()
    active_id = data.get("active_model_id")
    if not active_id:
        raise HTTPException(404, "尚未设置基座模型")
    for m in data["models"]:
        if m["id"] == active_id:
            return m
    raise HTTPException(404, "未找到激活模型配置")
```

启动：`uvicorn main:app --reload --port 8000`

const API_BASE = "http://127.0.0.1:8000";

// 前端内存中的状态
let state = { active_model_id: null, models: [] };

// ---------- 初始化 ----------
async function init() {
  try {
    const res = await fetch(`${API_BASE}/api/models`);
    state = await res.json();

    console.warn("后端未连接，使用空状态");
  }
  render();
}

// ---------- 渲染 ----------
function render() {
  renderModelList();
  renderBaseNode();

function renderModelList() {
  const box = document.getElementById("modelList");
  box.innerHTML = "";
  state.models.forEach(m => {
    const active = m.id === state.active_model_id;
    const card = document.createElement("div");
    card.className = "model-card" + (active ? " active" : "");
    card.draggable = true;
    card.ondragstart = e => e.dataTransfer.setData("text/plain", m.id);
    card.innerHTML = `
      <div class="actions">
        <button onclick="openModal('${m.id}')">✏️</button>
        <button onclick="removeModel('${m.id}')">🗑️</button>
      </div>
      <div class="name">${m.name} ${active ? '<span class="badge">已挂载</span>' : ''}</div>
      <div class="provider">${m.provider} · ${m.model_name}</div>
    `;
    box.appendChild(card);
  });
}

function renderBaseNode() {
  const mounted = document.getElementById("mountedArea");
  const active = state.models.find(m => m.id === state.active_model_id);
  if (!active) {
    mounted.innerHTML = `<div class="empty-slot">（空）暂无挂载模型</div>`;
    return;
  }
  mounted.innerHTML = `
    <div class="mounted-card">
      <button class="unplug-btn" onclick="unplug()">拔出 ✕</button>
      <div class="name">${active.name}</div>
      <div class="meta">${active.provider} · ${active.model_name}</div>
    </div>
  `;
}

// ---------- 拖拽 ----------
function onDragOver(e) {
  e.preventDefault();
  document.getElementById("baseNode").classList.add("dragover");
}
function onDragLeave(e) {
  document.getElementById("baseNode").classList.remove("dragover");
}
function onDrop(e) {
  e.preventDefault();
  document.getElementById("baseNode").classList.remove("dragover");
  const id = e.dataTransfer.getData("text/plain");

  render();

function unplug() {
  state.active_model_id = null;
  render();

function removeModel(id) {
  if (!confirm("确认删除该模型配置？")) return;
  state.models = state.models.filter(m => m.id !== id);
  if (state.active_model_id === id) state.active_model_id = null;

function openModal(id) {
  document.getElementById("mask").classList.add("show");
  if (id) {
    const m = state.models.find(x => x.id === id);
    document.getElementById("modalTitle").innerText = "编辑模型";
    document.getElementById("f_id").value = m.id;
    document.getElementById("f_name").value = m.name;
    document.getElementById("f_provider").value = m.provider;
    document.getElementById("f_api_base").value = m.api_base;
    document.getElementById("f_api_key").value = m.api_key;
    document.getElementById("f_model_name").value = m.model_name;
    document.getElementById("f_temperature").value = m.temperature;
    document.getElementById("f_max_tokens").value = m.max_tokens;
    document.getElementById("f_extra").value = JSON.stringify(m.extra_params || {});

    document.getElementById("modalTitle").innerText = "新增模型";

      .forEach(id => document.getElementById(id).value = "");
    document.getElementById("f_provider").value = "openai";
    document.getElementById("f_temperature").value = 0.7;
    document.getElementById("f_max_tokens").value = 4096;

}
function closeModal() {
  document.getElementById("mask").classList.remove("show");
}

function submitModal() {
  const id = document.getElementById("f_id").value || crypto.randomUUID();
  let extra = {};
  try {
    extra = JSON.parse(document.getElementById("f_extra").value || "{}");
  } catch (e) {
    alert("额外参数必须是合法 JSON");
    return;
  }
  const data = {
    id,
    name: document.getElementById("f_name").value.trim(),
    provider: document.getElementById("f_provider").value,
    api_base: document.getElementById("f_api_base").value.trim(),
    api_key: document.getElementById("f_api_key").value.trim(),
    model_name: document.getElementById("f_model_name").value.trim(),
    temperature: parseFloat(document.getElementById("f_temperature").value),
    max_tokens: parseInt(document.getElementById("f_max_tokens").value),
    extra_params: extra
  };
  if (!data.name || !data.model_name) {
    alert("名称和模型标识为必填");
    return;
  }
  const idx = state.models.findIndex(m => m.id === id);
  if (idx >= 0) {
    state.models[idx] = data;
  } else {
    state.models.push(data);
  }
  closeModal();
  render();
}

// ---------- 保存到后端 ----------
async function saveToBackend() {
  try {
    const res = await fetch(`${API_BASE}/api/models/save`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(state)
    });
    if (!res.ok) throw new Error(await res.text());
    const result = await res.json();
    state = result.data; // 服务端可能补全了新id
    render();
    showToast("保存成功 ✅");
  } catch (e) {
    showToast("保存失败：" + e.message);
  }
}

function showToast(msg) {
  const t = document.getElementById("toast");
  t.innerText = msg;
  t.classList.add("show");
  setTimeout(() => t.classList.remove("show"), 2000);
}

init();

## 四、真正的"可插拔"关键：适配器层（调用侧）

前面 UI + JSON 只是**配置管理**，真正解决"平台老换模型名和 API"的问题在于：**agent 的业务代码永远不直接 import 某个 SDK，而是通过一个统一接口拿 client**。建议在后端再加一个 `llm_factory.py`：

```python
# llm_factory.py
from main import load_data

class BaseLLMAdapter:
    def __init__(self, config: dict):
        self.config = config

    def chat(self, messages: list, **kwargs) -> str:
        raise NotImplementedError


class OpenAIAdapter(BaseLLMAdapter):
    def chat(self, messages, **kwargs):
        from openai import OpenAI
        client = OpenAI(api_key=self.config["api_key"], base_url=self.config["api_base"])
        resp = client.chat.completions.create(
            model=self.config["model_name"],
            messages=messages,
            temperature=self.config.get("temperature", 0.7),
            max_tokens=self.config.get("max_tokens", 4096),
            **self.config.get("extra_params", {})
        )
        return resp.choices[0].message.content


class AnthropicAdapter(BaseLLMAdapter):
    def chat(self, messages, **kwargs):
        import anthropic
        client = anthropic.Anthropic(api_key=self.config["api_key"])
        resp = client.messages.create(
            model=self.config["model_name"],
            max_tokens=self.config.get("max_tokens", 4096),
            messages=messages,
        )
        return resp.content[0].text


class CustomHTTPAdapter(BaseLLMAdapter):
    def chat(self, messages, **kwargs):
        import requests
        payload = {"model": self.config["model_name"], "messages": messages, **self.config.get("extra_params", {})}
        headers = {"Authorization": f"Bearer {self.config['api_key']}"}
        resp = requests.post(f"{self.config['api_base']}/chat/completions", json=payload, headers=headers)
        return resp.json()["choices"][0]["message"]["content"]


ADAPTER_MAP = {
    "openai": OpenAIAdapter,
    "azure_openai": OpenAIAdapter,   # base_url 换成 azure endpoint 即可复用
    "anthropic": AnthropicAdapter,
    "custom": CustomHTTPAdapter,
    # 智谱/通义/文心可各自补一个 Adapter，接口都遵循 OpenAI 兼容协议时可直接复用 OpenAIAdapter
}


def get_active_llm_client() -> BaseLLMAdapter:
    data = load_data()
    active_id = data.get("active_model_id")
    config = next((m for m in data["models"] if m["id"] == active_id), None)
    if not config:
        raise RuntimeError("尚未挂载基座模型，请先在页面配置")
    adapter_cls = ADAPTER_MAP.get(config["provider"], CustomHTTPAdapter)
    return adapter_cls(config)
```

以后 agent 主逻辑里永远只写：

```python
from llm_factory import get_active_llm_client

client = get_active_llm_client()
reply = client.chat([{"role": "user", "content": "你好"}])
```

**换平台/换模型名时，只需要在页面拖一下新模型、点保存，代码一行不用改**——这才是"可插拔基座"真正落地的地方，UI 和 JSON 只是让这个切换变得可视化、可管理。
