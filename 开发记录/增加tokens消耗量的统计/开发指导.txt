# Token 消耗统计与展示 — 设计方案

这个功能要解决三个问题：**① 从哪拿到 token 数据 ② 怎么在架构里传递这个数据（尤其是流式场景）③ 怎么落地存储用于后续统计**。下面按这个顺序设计。

---

## 一、先明确需求边界

| 场景 | 展示内容 |
|---|---|
| 非流式对话结束 | 本次请求的 prompt_tokens / completion_tokens / total_tokens |
| 流式对话结束 | 同上，但 token 数只有在流结束时才能拿到 |
| 单条消息展示 | 消息气泡下方一行小字 |


---

## 二、数据结构设计

```python
# schemas.py
from pydantic import BaseModel
from typing import Optional
from datetime import datetime


class TokenUsage(BaseModel):
    prompt_tokens: int = 0
    completion_tokens: int = 0
    total_tokens: int = 0
    model_id: Optional[str] = None
    model_name: Optional[str] = None
    provider: Optional[str] = None
    estimated_cost: Optional[float] = None   # 人民币/美元，按模型定价计算
    is_estimated: bool = False               # 标记：是真实值还是本地估算值


class ChatResult(BaseModel):
    content: str
    usage: TokenUsage
```

之所以把 `usage` 和 `content` 打包成 `ChatResult`，是因为**adapter 层的返回值类型需要统一改造**——原来只返回字符串，现在必须把 usage 一起带出来，这是本次改造的核心变化点。

---

## 三、模型定价配置（用于估算成本，可选但推荐）

在 `models.json` 的每个模型配置里加一个 `pricing` 字段：

```json
{
  "id": "xxx",
  "name": "线上-GPT4turbo",
  "provider": "openai",
  "model_name": "gpt-4-turbo",
  "pricing": {
    "input_per_1k": 0.01,
    "output_per_1k": 0.03,
    "currency": "USD"
  }
}
```

对应 Pydantic 模型加一个可选字段即可：

```python
class PricingConfig(BaseModel):
    input_per_1k: float = 0
    output_per_1k: float = 0
    currency: str = "USD"

# ModelConfig 里加：
pricing: Optional[PricingConfig] = None
```

## 四、Adapter 层改造：统一提取 usage

这是最关键的一步，各家 API 返回 usage 字段的格式都不同，需要在 adapter 内部"翻译"成统一格式。

```python
# llm_adapters.py（在原基础上改造）

from schemas import ChatResult, TokenUsage


def calc_cost(usage: TokenUsage, pricing: dict) -> float:
    if not pricing:
        return 0.0
    return round(
        usage.prompt_tokens / 1000 * pricing.get("input_per_1k", 0)
        + usage.completion_tokens / 1000 * pricing.get("output_per_1k", 0),
        6,
    )


class OpenAIAdapter(BaseLLMAdapter):
    async def chat(self, messages, **kwargs) -> ChatResult:
        resp = await self.client.chat.completions.create(
            model=self.config["model_name"],
            messages=messages,
            temperature=self.config.get("temperature", 0.7),
            max_tokens=self.config.get("max_tokens", 4096),
        )
        content = resp.choices[0].message.content
        usage = TokenUsage(
            prompt_tokens=resp.usage.prompt_tokens,
            completion_tokens=resp.usage.completion_tokens,
            total_tokens=resp.usage.total_tokens,
            model_id=self.config["id"],
            model_name=self.config["model_name"],
            provider="openai",
        )
        usage.estimated_cost = calc_cost(usage, self.config.get("pricing"))
        return ChatResult(content=content, usage=usage)

    async def chat_stream(self, messages, **kwargs):
        """
        流式场景：OpenAI 支持 stream_options={"include_usage": True}
        这样最后一个 chunk 会额外带一份 usage 数据
        """
        stream = await self.client.chat.completions.create(
            model=self.config["model_name"],
            messages=messages,
            stream=True,
            stream_options={"include_usage": True},
            temperature=self.config.get("temperature", 0.7),
        )
        async for chunk in stream:
            if chunk.choices and chunk.choices[0].delta.content:
                yield {"type": "content", "data": chunk.choices[0].delta.content}
            if chunk.usage:  # 只有最后一个 chunk 会带 usage
                usage = TokenUsage(
                    prompt_tokens=chunk.usage.prompt_tokens,
                    completion_tokens=chunk.usage.completion_tokens,
                    total_tokens=chunk.usage.total_tokens,
                    model_id=self.config["id"],
                    model_name=self.config["model_name"],
                    provider="openai",
                )
                usage.estimated_cost = calc_cost(usage, self.config.get("pricing"))
                yield {"type": "usage", "data": usage.dict()}


class AnthropicAdapter(BaseLLMAdapter):
    async def chat(self, messages, **kwargs) -> ChatResult:
        resp = await self.client.messages.create(
            model=self.config["model_name"],
            max_tokens=self.config.get("max_tokens", 4096),
            messages=messages,
        )
        usage = TokenUsage(
            prompt_tokens=resp.usage.input_tokens,
            completion_tokens=resp.usage.output_tokens,
            total_tokens=resp.usage.input_tokens + resp.usage.output_tokens,
            model_id=self.config["id"],
            model_name=self.config["model_name"],
            provider="anthropic",
        )
        usage.estimated_cost = calc_cost(usage, self.config.get("pricing"))
        return ChatResult(content=resp.content[0].text, usage=usage)

    async def chat_stream(self, messages, **kwargs):
        """
        Anthropic 的流式：usage 分散在 message_start（输入token）
        和 message_delta（输出token）两个事件里，需要累加
        """
        prompt_tokens = 0
        completion_tokens = 0
        async with self.client.messages.stream(
            model=self.config["model_name"],
            max_tokens=self.config.get("max_tokens", 4096),
            messages=messages,
        ) as stream:
            async for event in stream:
                if event.type == "message_start":
                    prompt_tokens = event.message.usage.input_tokens
                elif event.type == "content_block_delta":
                    yield {"type": "content", "data": event.delta.text}
                elif event.type == "message_delta":
                    completion_tokens = event.usage.output_tokens

        usage = TokenUsage(
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            total_tokens=prompt_tokens + completion_tokens,
            model_id=self.config["id"],
            model_name=self.config["model_name"],
            provider="anthropic",
        )
        usage.estimated_cost = calc_cost(usage, self.config.get("pricing"))
        yield {"type": "usage", "data": usage.dict()}
```

### 兜底方案：模型不返回 usage 怎么办？

有些自建/第三方接口压根不返回 usage 字段（`CustomHTTPAdapter` 常见）。这种情况需要**本地估算**，可以用简单的字符数近似，或者用 `tiktoken` 精确计算（如果是 OpenAI 系列 tokenizer 兼容）：

```python
def estimate_tokens(text: str) -> int:
    """粗略估算：中文约1.5字符/token，英文约4字符/token，这里取保守值"""
    try:
        import tiktoken
        enc = tiktoken.get_encoding("cl100k_base")
        return len(enc.encode(text))
    except Exception:
        # 兜底：中英混合场景下的经验值
        return max(1, len(text) // 2)


class CustomHTTPAdapter(BaseLLMAdapter):
    async def chat(self, messages, **kwargs) -> ChatResult:
        resp = await self._do_request(messages)
        data = resp.json()
        content = data["choices"][0]["message"]["content"]

        if "usage" in data:
            usage = TokenUsage(
                prompt_tokens=data["usage"]["prompt_tokens"],
                completion_tokens=data["usage"]["completion_tokens"],
                total_tokens=data["usage"]["total_tokens"],
                model_id=self.config["id"],
                model_name=self.config["model_name"],
                provider=self.config["provider"],
            )
        else:
            # 接口没给 usage，本地估算，并打上标记告诉前端这是估算值
            prompt_text = "".join(m["content"] for m in messages)
            usage = TokenUsage(
                prompt_tokens=estimate_tokens(prompt_text),
                completion_tokens=estimate_tokens(content),
                model_id=self.config["id"],
                model_name=self.config["model_name"],
                provider=self.config["provider"],
                is_estimated=True,
            )
            usage.total_tokens = usage.prompt_tokens + usage.completion_tokens

        usage.estimated_cost = calc_cost(usage, self.config.get("pricing"))
        return ChatResult(content=content, usage=usage)
```

`is_estimated=True` 这个标记很重要——前端展示的时候可以加一个"约"字或者标注"估算值"，避免用户误以为是精确计费数据。

## 五、LLMService 层：统一出口 + 落盘统计日志

```python
# llm_service.py
import json
import time
from datetime import datetime
from pathlib import Path
from schemas import ChatResult, TokenUsage

USAGE_LOG_FILE = "usage_log.jsonl"


class LLMService:
    async def chat(self, messages, conversation_id: str = None, **kwargs) -> ChatResult:
        config = registry.get_active_config()
        if not config:
            raise NoActiveModelError("尚未挂载基座模型")

        adapter = await self._get_adapter(config)
        start = time.time()
        result: ChatResult = await adapter.chat(messages, **kwargs)
        elapsed = round(time.time() - start, 2)

        self._log_usage(result.usage, conversation_id, elapsed)
        return result

    async def chat_stream(self, messages, conversation_id: str = None, **kwargs):
        config = registry.get_active_config()
        if not config:
            raise NoActiveModelError("尚未挂载基座模型")

        adapter = await self._get_adapter(config)
        start = time.time()
        async for event in adapter.chat_stream(messages, **kwargs):
            if event["type"] == "usage":
                usage = TokenUsage(**event["data"])
                elapsed = round(time.time() - start, 2)
                self._log_usage(usage, conversation_id, elapsed)
            yield event

    def _log_usage(self, usage: TokenUsage, conversation_id: str, elapsed: float):
        """落盘为 JSONL，后续可以直接用 pandas 读取做报表"""
        record = {
            "timestamp": datetime.now().isoformat(),
            "conversation_id": conversation_id,
            "elapsed_seconds": elapsed,
            **usage.dict(),
        }
        with open(USAGE_LOG_FILE, "a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
```

用 **JSONL（每行一条 JSON）** 而不是单个大 JSON 文件，是因为这是日志类数据、只追加不修改，JSONL 天然支持流式写入和后续用 `pandas.read_json(lines=True)` 直接读取分析，不会有并发写入损坏整个文件的风险。

## 六、API 接口改造

### 非流式接口

```python
@router.post("/api/agent/chat")
async def agent_chat(req: ChatRequest):
    result = await llm_service.chat(req.messages, conversation_id=req.conversation_id)
    return {
        "reply": result.content,
        "usage": result.usage.dict()
    }
```

### 流式接口（SSE）—— 关键：content 和 usage 用不同事件类型区分

```python
@router.post("/api/agent/chat_stream")
async def agent_chat_stream(req: ChatRequest):
    async def event_generator():
        async for event in llm_service.chat_stream(req.messages, conversation_id=req.conversation_id):
            if event["type"] == "content":
                yield f"event: content\ndata: {json.dumps({'text': event['data']}, ensure_ascii=False)}\n\n"
            elif event["type"] == "usage":
                yield f"event: usage\ndata: {json.dumps(event['data'], ensure_ascii=False)}\n\n"
        yield "event: done\ndata: {}\n\n"

    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

给 SSE 事件加 `event: content` / `event: usage` / `event: done` 类型区分，前端用 `EventSource` 或手写解析时可以精准分流处理，不需要猜测数据类型。

### 7.2 流式场景：用 fetch + ReadableStream 手动解析 SSE

```javascript
async function sendMessageStream(userText) {
  appendUserBubble(userText);
  const aiBubble = createEmptyAIBubble();  // 先创建一个空气泡，逐字填充

  const res = await fetch(`${API_BASE}/api/agent/chat_stream`, {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({ messages: buildMessages(userText), conversation_id: currentConvId })
  });

  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });

    const parts = buffer.split("\n\n");
    buffer = parts.pop(); // 最后一段可能不完整，留到下一次拼接

    for (const part of parts) {
      const eventMatch = part.match(/event: (\w+)/);
      const dataMatch = part.match(/data: (.+)/);
      if (!eventMatch || !dataMatch) continue;

      const eventType = eventMatch[1];
      const payload = JSON.parse(dataMatch[1]);

      if (eventType === "content") {
        appendTextToBubble(aiBubble, payload.text);
      } else if (eventType === "usage") {
        renderUsageBar(aiBubble, payload);   // 流结束时渲染 token 统计条
        accumulateSessionUsage(payload);
      }
    }
  }
}

function renderUsageBar(bubble, usage) {
  const bar = document.createElement("div");
  bar.className = "usage-bar";
  bar.innerHTML = `🔢 ${usage.prompt_tokens} + ${usage.completion_tokens} = ${usage.total_tokens} tokens` +
    (usage.estimated_cost ? ` · 约 $${usage.estimated_cost.toFixed(4)}` : "") +
    (usage.is_estimated ? " (估算值)" : "");
  bubble.appendChild(bar);
}
```

### 7.4 会话级累加（可选，放在对话窗口顶部）

```javascript
let sessionUsage = { prompt_tokens: 0, completion_tokens: 0, total_tokens: 0, cost: 0 };

function accumulateSessionUsage(usage) {
  sessionUsage.prompt_tokens += usage.prompt_tokens;
  sessionUsage.completion_tokens += usage.completion_tokens;
  sessionUsage.total_tokens += usage.total_tokens;
  sessionUsage.cost += usage.estimated_cost || 0;
  document.getElementById("sessionUsageBar").innerText =
    `本次会话共消耗 ${sessionUsage.total_tokens} tokens（约 $${sessionUsage.cost.toFixed(4)}）`;
}
```


## 九、这套设计的核心逻辑总结

1. **Adapter 层负责"翻译"**：把各平台千奇百怪的 usage 格式，统一转换成 `TokenUsage` 这一个标准结构，不支持 usage 的兜底用本地估算并打标记。
2. **流式场景的难点是"usage 只在最后出现"**，用 SSE 的 `event: usage` 单独区分，前端在收到这个事件时才渲染统计条，不影响逐字打字的显示体验。
3. **LLMService 是唯一的落盘点**，无论调用来自哪个业务模块，token 统计都会经过这一层被记录，保证统计口径统一、不会漏记。
