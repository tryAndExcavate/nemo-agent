## 一、整体思路梳理

### 1.1 与之前方案的核心差异

| 维度 | 之前方案 | 本次方案 |
|---|---|---|
| 触发时机 | 每次对话都跑一遍裁剪逻辑 | **只有当前上下文占用达到阈值（如60%）才触发**，平时走快速通道直接透传全部历史 |
| 相关性召回 | Embedding余弦相似度 | **BM25 / 关键词重叠**，零API成本，纯本地计算 |
| 检索范围 | （未细分） | 明确限定**会话内**，候选池是当前conversation_id的历史消息 |

### 1.2 完整流程图

```
每轮对话请求进来
   ↓
【触发判断】计算当前 system_prompt + 已有摘要 + 未覆盖历史 + 当前输入 的总token数
   ↓
   占用率 < 阈值(如60%)
   ├─ 是 → 快速通道：直接拼接全部历史原文，跳过BM25/摘要计算，零额外开销
   └─ 否 → 进入三层压缩：
            ↓
        Tier 1 近期窗口：从最新消息往前，贪心装填到预算内，保留原文
            ↓
        Tier 2 BM25召回：在"未被摘要覆盖 且 未被Tier1收录"的历史里，
                          用BM25对当前用户输入做检索，取Top-K相关消息，保留原文
            ↓
        Tier 3 摘要压缩：Tier1之外、被预算挤出的部分，增量摘要
                          （旧摘要+新裁掉的消息 → 合并新摘要，写回缓存）
            ↓
        组装最终消息数组，返回
```

**触发阈值卡在60%左右**是经验值——留出40%缓冲，既不会太频繁触发压缩（增加延迟），也不会等到99%满了才压缩导致有些边界case算错就直接超限报错。

### 3.1 Token计数器（复用之前设计，保持不变）

```python
# token_counter.py
import tiktoken


class TokenCounter:
    _encoder = tiktoken.get_encoding("cl100k_base")

    @classmethod
    def count(cls, text: str, provider: str = "openai") -> int:
        if not text:
            return 0
        if provider in ("openai", "azure_openai"):
            return len(cls._encoder.encode(text))
        # 非OpenAI系列的粗略估算，保守偏高，防止实际超预算
        cjk_count = sum(1 for ch in text if '\u4e00' <= ch <= '\u9fff')
        other_count = len(text) - cjk_count
        return int(cjk_count / 1.3 + other_count / 3.5)

    @classmethod
    def count_messages(cls, messages: list, provider: str = "openai") -> int:
        total = sum(cls.count(m["content"], provider) for m in messages)
        return total + len(messages) * 4  # 每条消息的角色/格式开销，经验值
```

### 3.2 触发判断器（本次新增的核心组件）

```python
# compression_trigger.py
from dataclasses import dataclass
from typing import List, Dict, Optional
from token_counter import TokenCounter


@dataclass
class TriggerResult:
    should_compress: bool
    used_tokens: int
    budget_tokens: int
    usage_ratio: float


class CompressionTrigger:
    """
    只有当上下文占用达到阈值时才触发压缩逻辑，
    平时走快速通道，避免每轮对话都跑BM25+摘要的额外开销
    """

    def __init__(self, trigger_ratio: float = 0.7):
        self.trigger_ratio = trigger_ratio  # 达到这个占用比例才触发压缩

    def check(
        self,
        history: List[Dict],
        system_prompt: str,
        current_input: str,
        existing_summary: Optional[Dict],
        model_config: dict,
    ) -> TriggerResult:
        provider = model_config.get("provider", "openai")
        max_window = model_config.get("max_context_window", 8000)
        reserve_for_reply = model_config.get("reserve_for_reply", 1000)
        budget = int((max_window - reserve_for_reply) * 0.9)  # 安全余量打9折

        used = TokenCounter.count(system_prompt, provider)
        used += TokenCounter.count(current_input, provider)
        if existing_summary:
            used += existing_summary["token_count"]

        # 只统计"尚未被摘要覆盖"的历史消息（已覆盖的历史不重复计入原文预算）
        covered_id = existing_summary["covered_up_to_message_id"] if existing_summary else 0
        uncovered_history = [m for m in history if m["id"] > covered_id]
        used += TokenCounter.count_messages(uncovered_history, provider)

        ratio = used / budget if budget > 0 else 1.0
        return TriggerResult(
            should_compress=ratio >= self.trigger_ratio,
            used_tokens=used,
            budget_tokens=budget,
            usage_ratio=round(ratio, 3),
        )
```

### 3.3 BM25 相关性召回器（本次核心新增）

```python
# bm25_retriever.py
import re
import jieba
from typing import List, Dict, Set
from rank_bm25 import BM25Okapi
from token_counter import TokenCounter

# 常见中英文停用词，过滤掉没有信息量的高频词，提升BM25区分度
STOPWORDS = {
    "的", "了", "是", "在", "我", "你", "他", "她", "它", "这", "那", "和", "与",
    "就", "都", "而", "及", "或", "一个", "没有", "我们", "你们", "他们", "自己",
    "什么", "怎么", "为什么", "吗", "呢", "啊", "吧", "the", "a", "an", "is",
    "are", "was", "were", "to", "of", "and", "in", "on", "for", "it", "this",
}


def tokenize(text: str) -> List[str]:
    """中英文混合分词：中文用jieba切词，英文按单词切分，统一转小写并过滤停用词"""
    text = text.lower()
    # 先用jieba对中文部分分词（jieba对英文/数字也能兼容切分）
    words = jieba.lcut(text)
    tokens = []
    for w in words:
        w = w.strip()
        if not w or w in STOPWORDS:
            continue
        # 过滤纯标点符号
        if re.fullmatch(r"[^\w]+", w):
            continue
        tokens.append(w)
    return tokens


class BM25Retriever:
    """
    会话内相关性召回：候选池是当前conversation的历史消息，
    每次触发压缩时临时构建BM25索引（会话内消息量通常几十到几百条，现场构建性能足够）
    """

    def __init__(self, top_k: int = 5, min_score: float = 1.0):
        self.top_k = top_k
        self.min_score = min_score  # BM25分数没有固定上限，这个阈值需要结合实际数据调参

    def retrieve(
        self,
        candidates: List[Dict],       # 候选消息列表，每条含 id/role/content
        query_text: str,
        exclude_ids: Set[int],        # 已经在Tier1近期窗口里的消息，检索时排除
        token_budget: int,
        provider: str = "openai",
    ) -> List[Dict]:
        pool = [c for c in candidates if c["id"] not in exclude_ids and len(c["content"].strip()) >= 4]
        if not pool:
            return []

        corpus_tokens = [tokenize(c["content"]) for c in pool]
        # 空分词结果的文档BM25库处理不了，过滤掉
        valid = [(c, toks) for c, toks in zip(pool, corpus_tokens) if toks]
        if not valid:
            return []

        pool = [v[0] for v in valid]
        corpus_tokens = [v[1] for v in valid]

        bm25 = BM25Okapi(corpus_tokens)
        query_tokens = tokenize(query_text)
        if not query_tokens:
            return []

        scores = bm25.get_scores(query_tokens)

        scored = [
            (score, msg) for score, msg in zip(scores, pool) if score >= self.min_score
        ]
        scored.sort(key=lambda x: -x[0])

        # 按相关度从高到低贪心装填，直到达到token预算或数量上限
        selected, used_tokens = [], 0
        for score, msg in scored[: self.top_k * 3]:  # 多取一些候选，再按预算精筛
            t = TokenCounter.count(msg["content"], provider)
            if used_tokens + t > token_budget or len(selected) >= self.top_k:
                break
            used_tokens += t
            selected.append({**msg, "relevance_score": round(float(score), 3)})

        return selected
```

### 3.4 三层压缩器整合（主体）

```python
# context_compressor.py
import asyncio
import logging
from dataclasses import dataclass, field
from typing import List, Dict, Optional

from token_counter import TokenCounter
from compression_trigger import CompressionTrigger
from bm25_retriever import BM25Retriever

logger = logging.getLogger("context_compressor")


@dataclass
class CompressedContext:
    messages: List[Dict]
    was_compressed: bool                       # 本轮是否触发了完整的三层压缩
    dropped_message_count: int = 0
    recalled_message_ids: List[int] = field(default_factory=list)
    usage_ratio: float = 0.0                    # 触发前的占用率，便于监控/前端展示


class ContextCompressor:
    def __init__(
        self,
        db_session_factory,
        summarizer,               # 复用你的LLMService，专门调"摘要模型"
        trigger_ratio: float = 0.7,
        bm25_top_k: int = 5,
    ):
        self.db_session_factory = db_session_factory
        self.summarizer = summarizer
        self.trigger = CompressionTrigger(trigger_ratio=trigger_ratio)
        self.retriever = BM25Retriever(top_k=bm25_top_k)

    async def compress(
        self,
        conversation_id: int,
        system_prompt: str,
        current_input: str,
        model_config: dict,
    ) -> CompressedContext:
        provider = model_config.get("provider", "openai")

        history = await self._load_history(conversation_id)
        existing_summary = await self._load_summary(conversation_id)

        trigger_result = self.trigger.check(
            history, system_prompt, current_input, existing_summary, model_config
        )

        # ---- 快速通道：占用率没到阈值，直接透传全部历史，不跑BM25/摘要 ----
        if not trigger_result.should_compress:
            messages = [{"role": "system", "content": system_prompt}]
            if existing_summary:
                messages.append({
                    "role": "system",
                    "content": f"[更早对话的背景摘要]\n{existing_summary['summary_text']}"
                })
            covered_id = existing_summary["covered_up_to_message_id"] if existing_summary else 0
            for m in history:
                if m["id"] > covered_id:
                    messages.append({"role": m["role"], "content": m["content"]})
            messages.append({"role": "user", "content": current_input})

            return CompressedContext(
                messages=messages,
                was_compressed=False,
                usage_ratio=trigger_result.usage_ratio,
            )

        # ---- 触发完整三层压缩 ----
        logger.info(
            f"[ContextCompressor] conversation={conversation_id} "
            f"占用率{trigger_result.usage_ratio}达到阈值，触发压缩"
        )
        return await self._full_compress(
            conversation_id, system_prompt, current_input, model_config,
            history, existing_summary, trigger_result,
        )

    async def _full_compress(
        self, conversation_id, system_prompt, current_input, model_config,
        history, existing_summary, trigger_result,
    ) -> CompressedContext:
        provider = model_config.get("provider", "openai")
        budget = trigger_result.budget_tokens

        fixed_cost = TokenCounter.count(system_prompt, provider) + TokenCounter.count(current_input, provider)
        if existing_summary:
            fixed_cost += existing_summary["token_count"]
        remaining_budget = max(budget - fixed_cost, 0)

        covered_id = existing_summary["covered_up_to_message_id"] if existing_summary else 0
        uncovered = [m for m in history if m["id"] > covered_id]

        # 预算三分：近期窗口60% / BM25召回25% / 剩余给摘要自然消耗
        recency_budget = int(remaining_budget * 0.6)
        relevance_budget = int(remaining_budget * 0.25)

        # ---- Tier 1: 近期窗口，从最新往前贪心装填 ----
        kept, used, cut_index = [], 0, len(uncovered)
        for i in range(len(uncovered) - 1, -1, -1):
            t = TokenCounter.count(uncovered[i]["content"], provider)
            if used + t > recency_budget:
                cut_index = i + 1
                break
            used += t
            kept.insert(0, uncovered[i])
        else:
            cut_index = 0
        to_compress = uncovered[:cut_index]
        exclude_ids = {m["id"] for m in kept}

        # ---- Tier 2 (BM25召回) 和 Tier 3 (摘要生成) 并行执行，减少额外延迟 ----
        relevant_task = asyncio.create_task(
            asyncio.to_thread(   # BM25是CPU密集型同步计算，丢到线程池不阻塞事件循环
                self.retriever.retrieve,
                uncovered, current_input, exclude_ids, relevance_budget, provider,
            )
        )
        summary_task = (
            asyncio.create_task(self._generate_summary(
                existing_summary["summary_text"] if existing_summary else None, to_compress
            ))
            if to_compress else None
        )

        relevant_messages = await relevant_task
        final_summary_text = existing_summary["summary_text"] if existing_summary else None
        was_compressed = False
        if summary_task:
            final_summary_text = await summary_task
            await self._save_summary(conversation_id, final_summary_text, to_compress[-1]["id"], model_config)
            was_compressed = True

        # ---- 组装最终消息 ----
        messages = [{"role": "system", "content": system_prompt}]
        if final_summary_text:
            messages.append({
                "role": "system",
                "content": f"[以下是此对话更早历史的背景摘要，可能存在信息压缩]\n{final_summary_text}"
            })
        if relevant_messages:
            recall_text = "\n".join(
                f"- [{m['role']}]（相关度{m['relevance_score']}）: {m['content']}"
                for m in relevant_messages
            )
            messages.append({
                "role": "system",
                "content": f"[以下是与当前问题相关的历史原文片段，可作为精确依据引用]\n{recall_text}"
            })
        for m in kept:
            messages.append({"role": m["role"], "content": m["content"]})
        messages.append({"role": "user", "content": current_input})

        return CompressedContext(
            messages=messages,
            was_compressed=was_compressed,
            dropped_message_count=len(to_compress),
            recalled_message_ids=[m["id"] for m in relevant_messages],
            usage_ratio=trigger_result.usage_ratio,
        )

    async def _generate_summary(self, old_summary: Optional[str], new_messages: List[Dict]) -> str:
        conversation_text = "\n".join(f"{m['role']}: {m['content']}" for m in new_messages)
        if old_summary:
            prompt = (
                f"这是之前对话的摘要：\n{old_summary}\n\n"
                f"以下是新增的对话内容：\n{conversation_text}\n\n"
                f"请合并生成一份更新后的摘要（300字以内），保留关键事实、数字、用户明确提出的要求。"
            )
        else:
            prompt = (
                f"请将以下对话浓缩为一份摘要（300字以内），保留关键事实、数字、用户明确提出的要求：\n\n"
                f"{conversation_text}"
            )
        try:
            result = await self.summarizer.chat_with_utility_model(
                messages=[{"role": "user", "content": prompt}]
            )
            return result.content
        except Exception as e:
            logger.error(f"摘要生成失败，降级返回旧摘要或空: {e}")
            return old_summary or "（摘要生成失败，早期对话内容暂不可用）"

    # ---- 对接你现有MySQL持久化层，按实际情况实现 ----
    async def _load_history(self, conversation_id: int) -> List[Dict]:
        raise NotImplementedError

    async def _load_summary(self, conversation_id: int) -> Optional[Dict]:
        raise NotImplementedError

    async def _save_summary(self, conversation_id, summary_text, covered_id, model_config):
        raise NotImplementedError
```

## 四、MySQL 表结构（沿用之前的设计，未变）

```sql
CREATE TABLE context_summaries (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    conversation_id BIGINT NOT NULL,
    summary_text TEXT NOT NULL,
    covered_up_to_message_id BIGINT NOT NULL,
    token_count INT NOT NULL,
    summary_model_id VARCHAR(64) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uniq_conversation (conversation_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

**本方案不需要额外的embedding表**，比之前的embedding方案少了一整套"写时计算向量+存储"的基础设施，BM25索引是压缩触发时**现场构建**的，不需要持久化索引结构，这是选择BM25的一大工程简化优势。

## 六、参数调优建议表

| 参数 | 默认值 | 调优方向 |
|---|---|---|
| `trigger_ratio` | 0.6 | 越低触发越频繁（更省心但更多摘要调用成本），越高越省成本但风险窗口更小，建议先用0.7观察 |
| `bm25_top_k` | 5 | 召回条数太多会稀释近期窗口的预算，太少可能漏掉关键历史，建议3-8之间 |
| `min_score` | 1.0 | BM25分数没有固定量纲，这个值**必须用你实际业务的对话数据跑几十条case后再定**，不同语料库分数分布差异很大 |
| 预算三分比例（60/25/剩余） | Tier1:60% Tier2:25% | 如果你的场景对话轮次多但话题跳跃小，可以适当调高Tier1占比；如果经常"绕回旧话题"，调高Tier2占比 |


## 七、这套设计相比之前方案的收益总结

1. **零API成本**：BM25/关键词检索是纯本地计算，不需要额外调用embedding接口，也不用维护向量存储表
2. **触发阈值控制**：短对话完全不会触碰压缩逻辑，性能上跟"什么都不做"几乎一样，只有真正对话变长时才付出压缩的计算代价
3. **并行化**：BM25检索（CPU计算）和摘要生成（网络IO调用模型）并行跑，额外延迟被压到最低
4. **降级链路清晰**：BM25检索本身失败/空结果不影响主流程（顶多召回为空），摘要生成失败会返回旧摘要兜底，两层互不阻塞
